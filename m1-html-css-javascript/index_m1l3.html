01Ejecución de JavaScript

Inclusión en un documento HTML

<!DOCTYPE html>
<html lang="es">
<head>
  <meta charset="utf-8">
  <meta name="viewport" content="width=device-width, initial-scale=1">
  <title>Ejemplo de JavaScript</title>
  <script src="app.js" defer></script>
</head>
<body>
  <h1>JavaScript</h1>
</body>
</html>

app.js: salida en consola

console.log("Hola, JavaScript");

Mensaje modal del navegador

alert("Hola, JavaScript");

Ejecución con Node.js

node app.js

console.log funciona en el navegador y en Node.js. alert es una API del navegador, no está disponible por defecto en Node.js.

02var, let y const

var: ámbito de función

function ejemplo() {
  var numero = 5;
  console.log(numero);
  if (true) {
    var numero = 10;
    console.log(numero);
  }
  console.log(numero);
}
ejemplo();

let: ámbito de bloque

function ejemplo() {
  let numero = 5;
  console.log(numero);
  if (true) {
    let numero = 10;
    console.log(numero);
  }
  console.log(numero);
}
ejemplo();

const: sin reasignación

function ejemplo() {
  const numero = 5;
  console.log(numero);
  if (true) {
    const numero = 10;
    console.log(numero);
  }
  console.log(numero);
}
ejemplo();

La lectura recomienda camelCase para variables y funciones, PascalCase para clases y constructores, y nombres como MAX_INTENTOS para constantes de configuración. JavaScript distingue mayúsculas y minúsculas.

03Operadores

Aritméticos

let primerNumero = 10;
let segundoNumero = 5;
console.log(primerNumero + segundoNumero);
console.log(primerNumero * segundoNumero);
console.log(primerNumero % segundoNumero);

Otros operadores citados: resta -, división /, incremento ++ y decremento --.

Asignación

let numero = 10;
numero += 5;
console.log(numero);

Asignaciones compuestas citadas: +=, -=, *=, /= y %=.

Comparación

let edad = 25;
console.log(edad >= 18);
console.log(edad == '25');
console.log(edad === '25');

== puede convertir tipos; === compara tipo y valor. También se citan !=, !==, >, < y <=.

Lógicos

let esMayorDeEdad = true;
let tieneLicencia = false;
console.log(esMayorDeEdad && tieneLicencia);
console.log(esMayorDeEdad || tieneLicencia);
console.log(!esMayorDeEdad);

Ternario

let edad = 20;
let esMayorDeEdad = (edad >= 18) ? "Mayor de edad" : "Menor de edad";
console.log(esMayorDeEdad);

typeof

let nombre = "Juan";
console.log(typeof nombre);

Otros operadores citados: + para concatenar, . para propiedades, [] para índices, new para crear instancias y instanceof para consultar su cadena de prototipos.

05Condicionales

if

let edad = 18;
if (edad >= 18) {
  console.log("Es un adulto");
}

if...else

let edad = 16;
if (edad >= 18) {
  console.log("Es un adulto");
} else {
  console.log("Es menor de edad");
}

Aplicación complementaria de else if y switch

const edad = 16;
if (edad >= 18) {
  console.log("Adulto");
} else if (edad >= 13) {
  console.log("Adolescente");
} else {
  console.log("Niño");
}
const servicio = "SEO";
switch (servicio) {
  case "SEO":
    console.log("Optimización para motores de búsqueda");
    break;
  default:
    console.log("Otro servicio");
}

06Estructuras repetitivas

for

for (let indice = 0; indice < 5; indice++) {
  console.log("Número de iteración: " + indice);
}

while

let contador = 0;
while (contador < 5) {
  console.log("El contador es: " + contador);
  contador++;
}

do...while

let contador = 0;
do {
  console.log("El contador es: " + contador);
  contador++;
} while (contador < 5);

do...while ejecuta el bloque al menos una vez. En los tres ejemplos se obtienen cinco iteraciones, de 0 a 4.

07Funciones

Declaración y return

function calcularTotal(precio, cantidad) {
  return precio * cantidad;
}
let total = calcularTotal(10, 3);
console.log(total);

Expresión de función

let calcularDescuento = function(precio, descuento) {
  return precio - (precio * descuento / 100);
};
let precioConDescuento = calcularDescuento(100, 10);
console.log(precioConDescuento);

Función de flecha

let calcularImpuesto = (precio, tasa) => {
  return precio * tasa;
};
console.log(calcularImpuesto(100, 0.15));

Las funciones son valores de primera clase: pueden asignarse, pasarse como argumentos y devolverse desde otra función. Las flechas no tienen this ni arguments propios y no se pueden invocar con new.

08La palabra clave this

En un método, this depende de cómo se invoca la función. Con new se refiere a la nueva instancia. Los ejemplos siguientes aplican ambos casos.

09Objetos y métodos

Objeto simple

Objeto con método

let producto = {

  nombre: "Camisa",
  precio: 5000,
  cantidad: 100,
  mostrarDetalles: function() {
    return `${this.nombre} cuesta $${this.precio} y hay ${this.cantidad} en stock.`;
  }

console.log(producto.mostrarDetalles());

10Constructores y prototype

function Producto(nombre, precio, cantidad) {
  this.nombre = nombre;
  this.precio = precio;
  this.cantidad = cantidad;
}
Producto.prototype.mostrarDetalles = function() {
  return `${this.nombre} cuesta $${this.precio} y hay ${this.cantidad} en stock.`;
};
let producto1 = new Producto("Camisa", 5000, 100);
let producto2 = new Producto("Pantalones", 4000, 50);
console.log(producto1.mostrarDetalles());
console.log(producto2.mostrarDetalles());
console.log(producto1.mostrarDetalles === producto2.mostrarDetalles);
console.log(producto1 instanceof Producto);

Las instancias comparten el método del prototipo. Las dos comprobaciones finales son complementarias y muestran esa reutilización.

11Object.create()

let productoPrototipo = {
  mostrarDetalles: function() {
    return `${this.nombre} cuesta $${this.precio} y hay ${this.cantidad} en stock.`;
  }
};
let producto3 = Object.create(productoPrototipo);
producto3.nombre = "Zapatos";
producto3.precio = 6000;
producto3.cantidad = 30;
console.log(producto3.mostrarDetalles());

El nuevo objeto hereda de productoPrototipo. La cadena de prototipos termina en null; Object.prototype no es el mismo valor que null.

Acceso, modificación y recorrido

let ropa = ["Camisa", "Pantalón", "Remera", 25, true];
console.log(ropa[0]);
ropa[2] = "Campera";
console.log(ropa);

  console.log(ropa[indice]);

Constructor Array: un argumento numérico

const ropa = new Array(2);
console.log(ropa.length);
console.log(ropa[0]);

Constructor Array: varios elementos

const ropa = new Array("Camisa", "Pantalon");
console.log(ropa.length);

Método push

let miArreglo = [1, 2, 3];
miArreglo.push(4);
console.log(miArreglo);

La lectura también menciona pop, map y filter. Los índices empiezan en 0; new Array(2) crea dos posiciones vacías, no un arreglo que contiene el número 2.

13Herencia prototípica

Un objeto como prototipo

const producto = { nombre: "Camisa", precio: 5000 };
const oferta = Object.create(producto);
oferta.descuento = 1000;
console.log(oferta.nombre);
console.log(oferta.precio);
console.log(oferta.descuento);

Herencia mediante funciones constructoras

function Producto(nombre, precio) {
  this.nombre = nombre;
  this.precio = precio;
}
function Oferta(nombre, precio, descuento) {
  Producto.call(this, nombre, precio);
  this.descuento = descuento;
}
Oferta.prototype = Object.create(Producto.prototype);
Oferta.prototype.constructor = Oferta;
const oferta = new Oferta("Camisa", 5000, 1000);
console.log(oferta.nombre);
console.log(oferta.precio);
console.log(oferta.descuento);

14Clases y extends

Estructura citada

class NombreDeLaClase {
  constructor(parametros) {
  }
  metodo() {
  }
}

Producto y Oferta

class Producto {
  constructor(nombre, precio) {
    this.nombre = nombre;
    this.precio = precio;
  }
}
class Oferta extends Producto {
  constructor(nombre, precio, descuento) {
    super(nombre, precio);
    this.descuento = descuento;
  }
}
const oferta = new Oferta("Camisa", 5000, 1000);
console.log(oferta.nombre);
console.log(oferta.precio);
oferta.nombre = "Pantalón";
console.log(oferta.nombre);

extends establece herencia y super() inicializa la parte heredada. const impide reasignar la referencia, pero no impide modificar las propiedades del objeto.

JavaScript: app.js

class Servicio {
  constructor(nombre, precio, duracion) {
    this.nombre = nombre;
    this.precio = precio;
    this.duracion = duracion;
  }
  obtenerDetalles() {
    return `Precio: $${this.precio} | Duración: ${this.duracion}`;
  }
  crearElemento() {
    const articulo = document.createElement("article");
    const titulo = document.createElement("h3");
    const detalles = document.createElement("p");
    titulo.textContent = this.nombre;
    detalles.textContent = this.obtenerDetalles();
    articulo.append(titulo, detalles);
    return articulo;
  }
}
const servicios = [
  new Servicio("Gestión de Redes Sociales", 100000, "1 mes"),
  new Servicio("SEO", 300000, "2 meses"),
  new Servicio("Marketing de Contenidos", 150000, "1 mes"),
  new Servicio("Diseño y Desarrollo Web", 500000, "3 meses")
];
const contenedor = document.getElementById("lista-servicios");
for (const servicio of servicios) {
  contenedor.append(servicio.crearElemento());
}
