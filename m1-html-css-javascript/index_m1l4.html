04Carga de scripts

En la cabecera con defer

<head>
  <script src="script.js" defer></script>
</head>

Descarga el script sin bloquear el análisis del HTML y lo ejecuta después de terminarlo, antes de DOMContentLoaded.

En la cabecera con async

<head>
  <script src="script.js" async></script>
</head>

Lo ejecuta cuando termina la descarga; no garantiza el orden entre scripts ni que el DOM esté listo.

Al final del cuerpo

<body>
  <h1>Mi página</h1>
  <script src="script.js"></script>
</body>

  <h1>Bienvenido a la Página</h1>
  <div id="contenido">
    <p>Este es un párrafo original</p>
  </div>
  <button id="mostrarAlert">Mostrar Alerta</button>
  <button id="agregarElemento">Agregar Elemento</button>
  <button id="cambiarContenido">Cambiar Contenido</button>
</body>
</html>

JavaScript

window.onload = function() {
  window.alert('Comienzo de la página');
  const mostrarAlertButton = document.getElementById('mostrarAlert');
  const agregarElementoButton = document.getElementById('agregarElemento');
  const cambiarContenidoButton = document.getElementById('cambiarContenido');
  mostrarAlertButton.onclick = function() {
    window.alert('Se ha presionado el botón de mostrar alerta');
  };
  agregarElementoButton.onclick = function() {
    const nuevoElemento = document.createElement('h2');
    const texto = document.createTextNode('Este es un nuevo contenido');
    nuevoElemento.appendChild(texto);
    document.body.appendChild(nuevoElemento);
  };
  cambiarContenidoButton.onclick = function() {
    const contenidoDiv = document.getElementById('contenido');
    contenidoDiv.innerHTML = '<p>Se cambió el contenido</p>';
  };
};

13addEventListener

JavaScript: app.js

class Servicio {
  constructor(nombre, precio, duracion) {
    this.nombre = nombre;
    this.precio = precio;
    this.duracion = duracion;
  }
  obtenerDetalles() {
    return `Precio: $${this.precio} | Duración: ${this.duracion}`;
  }
  crearElemento() {
    const articulo = document.createElement('article');
    articulo.className = 'servicio';
    const titulo = document.createElement('h3');
    const detalles = document.createElement('p');
    titulo.textContent = this.nombre;
    detalles.textContent = this.obtenerDetalles();
    articulo.append(titulo, detalles);
    return articulo;
  }
}
document.addEventListener('DOMContentLoaded', function() {
  console.log('Página cargada: DOMContentLoaded');
  const servicios = [
    new Servicio('Gestión de Redes Sociales', 100000, '1 mes'),
    new Servicio('SEO', 300000, '2 meses'),
    new Servicio('Marketing de Contenidos', 150000, '1 mes'),
    new Servicio('Diseño y Desarrollo Web', 500000, '3 meses')
  ];
  const contenedor = document.getElementById('lista-servicios');
  for (const servicio of servicios) {
    const elemento = servicio.crearElemento();
    elemento.addEventListener('mouseover', function() {
      elemento.classList.add('destacado');
    });
    elemento.addEventListener('mouseout', function(evento) {
      if (!elemento.contains(evento.relatedTarget)) {
        elemento.classList.remove('destacado');
      }
    });
    contenedor.appendChild(elemento);
  }
  document.addEventListener('keydown', function(evento) {
    console.log('Tecla presionada:', evento.key);
  });
  const formulario = document.getElementById('formulario-contacto');
  formulario.addEventListener('submit', function(evento) {
    evento.preventDefault();
    alert('Formulario enviado');
  });
});
